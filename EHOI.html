<script>

    // Search functionality

    const searchInput = document.getElementById('search-input');

    const categoryFilter = document.getElementById('category-filter');

    const researchCards = document.querySelectorAll('.research-card');


    function filterResearch() {

        const searchTerm = searchInput.value.toLowerCase();

        const selectedCategory = categoryFilter.value;


        researchCards.forEach(card => {

            const title = card.querySelector('h2').textContent.toLowerCase();

            const description = card.querySelector('.description').textContent.toLowerCase();

            const tags = card.getAttribute('data-tags').toLowerCase();

            const category = card.getAttribute('data-category');


            const matchesSearch = title.includes(searchTerm) || 

                                description.includes(searchTerm) || 

                                tags.includes(searchTerm);

            

            const matchesCategory = !selectedCategory || category === selectedCategory;


            if (matchesSearch && matchesCategory) {

                card.style.display = 'block';

            } else {

                card.style.display = 'none';

            }

        });

    }


    searchInput.addEventListener('input', filterResearch);

    categoryFilter.addEventListener('change', filterResearch);


    // Add new research functionality

    function addNewResearch() {

        const title = prompt('Enter research title:');

        if (!title) return;


        const category = prompt('Enter category (science, technology, business, humanities, health):');

        if (!category) return;


        const description = prompt('Enter description:');

        if (!description) return;


        const tags = prompt('Enter tags (comma-separated):');

        if (!tags) return;


        const author = prompt('Enter author:');

        if (!author) return;


        const researchGrid = document.getElementById('research-grid');

        const newCard = document.createElement('div');

        newCard.className = 'research-card';

        newCard.setAttribute('data-category', category.toLowerCase());

        newCard.setAttribute('data-tags', tags.toLowerCase());


        const categoryClass = category.toLowerCase();

        const tagArray = tags.split(',').map(tag => tag.trim());


        newCard.innerHTML = `

            <span class="category-badge $${categoryClass}">$${category}</span>

            <h2>$${title}</h2>

            <p class="meta-info">Added: $${new Date().toLocaleDateString()} | Author: $${author}</p>

            <p class="description">$${description}</p>

            <div class="tags">

                ${tagArray.map(tag => `<span class="tag">${tag}</span>`).join('')}

            </div>

        `;


        researchGrid.appendChild(newCard);

        updateStats();

        filterResearch(); // Reapply filters

    }


    // Update statistics

    function updateStats() {

        const totalResearch = document.querySelectorAll('.research-card').length;

        const categories = new Set();

        const tags = new Set();


        document.querySelectorAll('.research-card').forEach(card => {

            categories.add(card.getAttribute('data-category'));

            const cardTags = card.getAttribute('data-tags').split(',');

            cardTags.forEach(tag => tags.add(tag.trim()));

        });


        document.getElementById('total-research').textContent = totalResearch;

        document.getElementById('categories-count').textContent = categories.size;

        document.getElementById('tags-count').textContent = tags.size;

    }


    // Initialize

    updateStats();

</script>
